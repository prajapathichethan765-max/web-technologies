            <h2>📘 What is a Variable?</h2>

            <p>
                A <span class="highlight">variable</span> is a named
                memory location that acts as a container to store
                data values during program execution.
            </p>

            <p>
                Every variable must be declared with a specific
                data type and a unique name before it can be used.
            </p>
        </section>

        <!-- Variable Types -->

        <section class="card" id="types">
            <h2>🚀 Types of Variables in Java</h2>

            <div class="variable-box">

                <div class="variable" id="local">
                    <h3>🐒 Local Variable</h3>

                    <p>
                        A variable declared inside a method,
                        constructor, or code block.
                    </p>

                    <div class="syntax">
                        datatype variableName = value;
                    </div>
                </div>

                <div class="variable" id="static">
                    <h3>🪿 Static Variable</h3>

                    <p>
                        A variable declared inside a class
                        using the static keyword. It is shared
                        among all objects of that class.
                    </p>

                    <div class="syntax">
                        static datatype variableName = value;
                    </div>
                </div>

                <div class="variable" id="parameter">
                    <h3>🦥 Parameter Variable</h3>

                    <p>
                        A variable declared inside a method
                        or constructor parameter list.
                    </p>

                    <div class="syntax">
                        datatype methodName(datatype variableName) { }
                    </div>
                </div>

            </div>
        </section>

        <!-- Local Variable Code -->

        <section class="card">
            <h2>💻 Program 1: Local Variable</h2>

            <p>
                Local variables are declared inside a method
                and can only be accessed within their scope.
            </p>

<pre>
public class Calculator {

    public void addNumbers() {

        int a = 10;
        int b = 20;

        int sum = a + b;

        System.out.println("Sum: " + sum);
    }

    public static void main(String[] args) {

        Calculator calc = new Calculator();

        calc.addNumbers();
    }
}
</pre>
        </section>

        <!-- Static Variable Code -->

        <section class="card">
            <h2>💻 Program 2: Static Variable</h2>

            <p>
                Static variables belong to the class and
                are shared by all objects of that class.
            </p>

<pre>
class Employee {

    String name;

    static int employeeCount = 0;

    public static final String COMPANY_NAME = "TechCorp";

    public Employee(String name) {

        this.name = name;

        employeeCount++;
    }

    public void displayInfo() {

        System.out.println("Name: " + name);

        System.out.println("Company: " + COMPANY_NAME);
    }

    public static void main(String[] args) {

        System.out.println("Initial count: " +
                           Employee.employeeCount);

        Employee emp1 = new Employee("Alice");

        Employee emp2 = new Employee("Bob");

        emp1.displayInfo();

        emp2.displayInfo();

        System.out.println("Total employees: " +
                           Employee.employeeCount);
    }
}
</pre>
        </section>

        <!-- Parameter Variable Code -->

        <section class="card">
            <h2>💻 Program 3: Parameter Variable</h2>

            <p>
                Parameter variables receive values when
                a method or constructor is called.
            </p>

<pre>
public class Employee {

    static String companyName = "TechCorp";

    String employeeName;

    double salary;

    public void setDetails(String name, double baseSalary) {

        this.employeeName = name;

        this.salary = baseSalary;
    }

    public void printTotalPay(double bonus) {

        double totalPay = this.salary + bonus;

        System.out.println("Company: " + companyName);

        System.out.println("Employee: " + employeeName);

        System.out.println("Total Pay: " + totalPay);
    }

    public static void main(String[] args) {

        Employee emp = new Employee();

        emp.setDetails("Alice", 75000.0);

        emp.printTotalPay(5000.0);
    }
}
</pre>
