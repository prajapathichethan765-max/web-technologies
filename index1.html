code for static veriable 
        class Employee {
    // Instance variable (each object gets its own copy)
    String name;
    
    // Static variable (shared across all objects)
    static int employeeCount = 0;
    
    // Static constant (common convention: public static final)
    public static final String COMPANY_NAME = "TechCorp";

    // Constructor
    public Employee(String name) {
        this.name = name;
        // Increment the shared static counter every time a new Employee is created
        employeeCount++; 
    }

    public void displayInfo() {
        System.out.println("Name: " + name + " | Company: " + COMPANY_NAME);
    }

    public static void main(String[] args) {
        // Access static variable before creating any objects
        System.out.println("Initial count: " + Employee.employeeCount); // Output: 0

        Employee emp1 = new Employee("Alice");
        Employee emp2 = new Employee("Bob");

        emp1.displayInfo();
        emp2.displayInfo();

        // Recommended way to access static variables: using the Class Name
        System.out.println("Total employees: " + Employee.employeeCount); // Output: 2
    }
}

        public class Employee {
    // 1. Class (Static) Variable: Shared by all Employee objects
    static String companyName = "TechCorp";

    // 2. Instance Variable: Unique to each Employee object
    String employeeName;
    double salary;

    // 3. Parameter Variables: 'name' and 'baseSalary' receive values when called
    public void setDetails(String name, double baseSalary) {
        this.employeeName = name;
        this.salary = baseSalary;
    }

    // 'bonus' is another Parameter Variable
    public void printTotalPay(double bonus) {
        // 4. Local Variable: Exists only inside printTotalPay()
        double totalPay = this.salary + bonus;

        System.out.println("Company: " + companyName);
        System.out.println("Employee: " + employeeName);
        System.out.println("Total Pay: $" + totalPay);
    }

    public static void main(String[] args) { // 'args' is a parameter variable
        Employee emp = new Employee();
        
        // Passing arguments ("Alice" and 75000.0) into the parameter variables
        emp.setDetails("Alice", 75000.0);
        
        // Passing argument (5000.0) into the 'bonus' parameter
        emp.printTotalPay(5000.0);
    }
}
